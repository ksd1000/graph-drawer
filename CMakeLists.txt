cmake_minimum_required(VERSION 3.10)
project(graph-drawer C)
if(NOT CMAKE_BUILD_TYPE)
	set(CMAKE_BUILD_TYPE Release)
	set(CBT_ISDEF " (default)")
else()
	set(CBT_ISDEF "")
endif()
message("-- ${PROJECT_NAME}->CMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}${CBT_ISDEF}")
message("-- ${PROJECT_NAME}->BUILD_SHARED_LIBS=${BUILD_SHARED_LIBS}")
add_subdirectory(expr)
add_subdirectory(graph)
set(CMAKE_C_FLAGS_RELEASE "-O3")
set(CMAKE_C_FLAGS_DEBUG "-O0 -g")
set(CMAKE_C_FLAGS_MINSIZEREL "-Os")
set(CMAKE_C_FLAGS_RELWITHDEBINFO "-O2 -g")
add_compile_options(-Wall -Wextra)
if(ASAN)
	add_link_options(-fsanitize=address,undefined)
endif()
set(LIBRARY_OUTPUT_PATH ${PROJECT_BINARY_DIR}/lib)
include_directories(expr)
include_directories(graph)
link_libraries(expr m)
add_library(common_symbols common_symbols.c)
add_definitions(-DCOMMON_SYMBOLS)
if(DEFINED CMAKE_HOST_UNIX)
	execute_process(COMMAND sh -c "sh ${PROJECT_SOURCE_DIR}/systable.sh >${PROJECT_BINARY_DIR}/systable.c")
	add_custom_target(expr.so.conf ALL sh -c \"sh "${PROJECT_SOURCE_DIR}/expr/lcdump" <"${PROJECT_SOURCE_DIR}/expr/expr.h" >"${PROJECT_BINARY_DIR}/expr.so.conf"\" BYPRODUCTS expr.so.conf SOURCES expr/expr.h)
	include_directories(${PROJECT_BINARY_DIR})
endif()
# so.conf for ltrace
macro(add_wave name type)
	add_executable(${name} wave.c)
	target_compile_definitions(${name} PRIVATE -DTEXT_ENABLED -D"${type}")
	target_link_libraries(${name} graph)
endmacro()

add_wave(wave S32LE)

add_executable(draw main.c)
target_link_libraries(draw graph common_symbols)

if(DEFINED CMAKE_HOST_UNIX)
	target_link_libraries(draw pthread)
endif()

add_executable(list list.c)
target_link_libraries(list common_symbols)
add_executable(calc calc.c)
target_link_libraries(calc common_symbols)

file(COPY echo.calc ping.calc uniq.calc DESTINATION ${PROJECT_BINARY_DIR})
